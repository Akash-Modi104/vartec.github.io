<?php
/* One-off helper: makes the hosting fetch static-site.zip from GitHub and unpack it into /htdocs.
   Only needed when you cannot upload the zip by hand in File Manager (Upload, then Unzip, does the same).

   1. Put the zip on the go-live-package branch and note the commit: the address below must contain the
      commit id, not the branch name, so the file cannot change under you.
   2. Fill in the three values. Use a new random key each time.
   3. In File Manager create a file with a random name in /htdocs (for example vt-h-x7k2m9.php) and paste this in.
   4. Open  https://vartec.global/<that name>?k=<key>&fetch=1   -> expect "ok":true and the number of pages.
      If you get the hosting's "Page Not Found", just open it again (its known PHP fault).
   5. Open  https://vartec.global/<that name>?k=<key>&last=1    -> the helper deletes itself. Check it is gone.
   Never leave this file on the server. */
$key  = 'PUT-A-RANDOM-KEY-HERE';
$from = 'https://raw.githubusercontent.com/Akash-Modi104/vartec.github.io/PUT-THE-COMMIT-ID-HERE/static-site.zip';
$sha  = 'PUT-THE-SHA-256-OF-THE-ZIP-HERE';

if (($_GET['k'] ?? '') !== $key) { http_response_code(404); exit; }
header('Content-Type: text/plain');
$dir = __DIR__;
$out = [];
if (isset($_GET['fetch'])) {
    // allow_url_fopen is off on this hosting, so use curl.
    $c = curl_init($from);
    curl_setopt_array($c, [CURLOPT_RETURNTRANSFER => true, CURLOPT_FOLLOWLOCATION => true, CURLOPT_TIMEOUT => 60]);
    $zip = curl_exec($c);
    $out['error'] = curl_error($c);
    curl_close($c);
    $out['bytes'] = is_string($zip) ? strlen($zip) : 0;
    if (is_string($zip) && hash('sha256', $zip) === $sha) {   // unpack only the exact file that was built
        file_put_contents("$dir/vt-s.zip", $zip);
        $a = new ZipArchive;
        if ($a->open("$dir/vt-s.zip") === true) { $out['ok'] = $a->extractTo($dir); $a->close(); }
        unlink("$dir/vt-s.zip");
    } else {
        $out['ok'] = false;
    }
}
$out['pages'] = count(glob("$dir/static-site/*/*/index.html"));
$out['home'] = is_file("$dir/static-site/index.html");
echo json_encode($out);
if (isset($_GET['last'])) { unlink(__FILE__); }
